package org.apache.tika.sax;

import org.apache.tika.sax.BasicContentHandlerFactory;
import org.apache.tika.parser.ParseContext;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import org.xml.sax.ContentHandler;
import java.io.OutputStream;
import java.io.OutputStreamWriter;
import java.io.UnsupportedEncodingException;
import java.nio.charset.Charset;
import java.util.Locale;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;
import org.xml.sax.helpers.DefaultHandler;
import org.apache.tika.annotation.TikaComponent;
import org.apache.tika.config.OutputLimits;

@ExtendWith(MockitoExtension.class)
public class BasicContentHandlerFactory_createHandler_4_0_Test {

    @Mock
    private ContentHandler formatHandler;

    @Mock
    private ParseContext parseContext;

    private BasicContentHandlerFactory factory;

    @BeforeEach
    public void setUp() {
        factory = new BasicContentHandlerFactory(BasicContentHandlerFactory.HANDLER_TYPE.BODY, 1000, true, parseContext);
    }

    @Test
    public void testCreateHandler() {
        ContentHandler handler = factory.createHandler();
        assertNotNull(handler);
        verify(formatHandler, times(1)).setThrowOnWriteLimitReached(true);
        verify(formatHandler, times(1)).setWriteLimit(1000);
        verify(formatHandler, times(1)).setParseContext(parseContext);
    }

    @Test
    public void testCreateHandlerWithIgnoreType() {
        factory = new BasicContentHandlerFactory(BasicContentHandlerFactory.HANDLER_TYPE.IGNORE, 1000, true, parseContext);
        ContentHandler handler = factory.createHandler();
        assertNotNull(handler);
        assertTrue(handler instanceof NoOpContentHandler);
    }

    @Test
    public void testCreateHandlerWithNoWriteLimit() {
        factory = new BasicContentHandlerFactory(BasicContentHandlerFactory.HANDLER_TYPE.BODY, -1, true, parseContext);
        ContentHandler handler = factory.createHandler();
        assertNotNull(handler);
        verify(formatHandler, times(1)).setThrowOnWriteLimitReached(true);
        verify(formatHandler, times(1)).setParseContext(parseContext);
    }

    @Test
    public void testCreateHandlerWithNoThrowOnWriteLimitReached() {
        factory = new BasicContentHandlerFactory(BasicContentHandlerFactory.HANDLER_TYPE.BODY, 1000, false, parseContext);
        assertThrows(IllegalArgumentException.class, () -> {
            factory.createHandler();
        });
    }

    @Test
    public void testCreateHandlerWithNullParseContext() {
        factory = new BasicContentHandlerFactory(BasicContentHandlerFactory.HANDLER_TYPE.BODY, 1000, false, null);
        assertThrows(IllegalArgumentException.class, () -> {
            factory.createHandler();
        });
    }
}
